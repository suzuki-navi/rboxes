### 2. Main Python Script (`server.py`)

**Structure:**
```
#!/usr/bin/env python3
# Imports
import http.server
import socketserver
import os
import sys
import json
from pathlib import Path

# Configuration Section
PORT = int(os.getenv('PORT', 8000))
HOST = os.getenv('HOST', '0.0.0.0')
SERVE_DIR = os.getenv('SERVE_DIR', '/workdir')

# Custom Handler Class
class CustomHTTPRequestHandler(http.server.SimpleHTTPRequestHandler):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, directory=SERVE_DIR, **kwargs)

    def log_message(self, format, *args):
        # Custom logging
        sys.stderr.write(f"[{self.date_time_string()}] {format % args}\n")

# Main Application Logic
def main():
    try:
        with socketserver.TCPServer((HOST, PORT), CustomHTTPRequestHandler) as httpd:
            print(f"Serving {SERVE_DIR} at http://{HOST}:{PORT}")
            httpd.serve_forever()
    except KeyboardInterrupt:
        print("\nServer stopped.")
        sys.exit(0)
    except Exception as e:
        print(f"Error: {e}", file=sys.stderr)
        sys.exit(1)

# Entry Point
if __name__ == "__main__":
    main()
```

### 3. Help Text (`help.txt`)
```
httpserver - Simple HTTP server for serving directories

Usage:
  httpserver [OPTIONS]

Environment Variables:
  PORT        Port to listen on (default: 8000)
  HOST        Host to bind to (default: 0.0.0.0)
  SERVE_DIR   Directory to serve (default: current directory)

Examples:
  httpserver                    # Serve current directory on port 8000
  PORT=3000 httpserver          # Serve on port 3000
  SERVE_DIR=/data httpserver    # Serve /data directory
```
